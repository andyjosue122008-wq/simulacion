<script>
  const form = document.getElementById('phishForm');
  const loginCard = document.getElementById('loginCard');
  const reveal = document.getElementById('revealCard');
  const restartBtn = document.getElementById('restartBtn');

  form.addEventListener('submit', function (e) {
    e.preventDefault(); // Nunca se envía nada a ningún servidor
    form.reset();        // Se borra inmediatamente lo escrito, no se guarda en ninguna variable
    loginCard.classList.add('hide');
    reveal.classList.add('show');
    reveal.scrollIntoView({ behavior: 'smooth', block: 'start' });
  });

  restartBtn.addEventListener('click', function () {
    reveal.classList.remove('show');
    loginCard.classList.remove('hide');
    window.scrollTo({ top: 0, behavior: 'smooth' });
  });
